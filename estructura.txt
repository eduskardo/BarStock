let mesasData = [];
let filtroActual = 'todas';
let busqueda = '';

document.addEventListener('DOMContentLoaded', () => {
    cargarMesas();
    initFiltros();
    initBuscador();
    initModal();

/* ---------- Cargar mesas desde la API ---------- */
async function cargarMesas() {
    try {
        mesasData = await MesasAPI.listar(filtroActual);
        renderMesas();
        actualizarContadores();
    } catch (e) {
        // Fallback demo si el backend no está levantado
        mesasData = getMesasDemo();
        renderMesas();
        actualizarContadores();
    }
}

/* ---------- Render del grid ---------- */
function renderMesas() {
    const grid = document.getElementById('mesasGrid');
    grid.innerHTML = '';

    const filtradas = mesasData.filter(m => {
        const matchFiltro = filtroActual === 'todas'
            || m.estado === filtroActual;
        const matchBusqueda = m.numero.toString().includes(busqueda)
            || (m.mesero || '').toLowerCase().includes(busqueda.toLowerCase());
        return matchFiltro && matchBusqueda;
    });

    if (!filtradas.length) {
        grid.innerHTML = `<p class="sin-resultados">No hay mesas que coincidan.</p>`;
        return;
    }

    filtradas.forEach(m => {
        const card = document.createElement('div');
        card.className = 'mesa-card';
        card.dataset.estado = m.estado;
        card.dataset.id = m.id;

        card.innerHTML = `
            <div class="mesa-numero">
                <span>#${m.numero}</span>
                <span class="estado-tag">${m.estado}</span>
            </div>
            <div class="mesa-info">
                <span>👤 ${m.mesero || 'Libre'}</span>
                <span>${m.comensales || 0} pax</span>
            </div>
            <div class="mesa-total">${Utils.formatearMoneda(m.total)}</div>
        `;

        card.addEventListener('click', () => abrirDetalleMesa(m.id));
        grid.appendChild(card);
    });
}

/* ---------- Filtros ---------- */
function initFiltros() {
    document.querySelectorAll('.filtro-btn').forEach(btn => {
        btn.addEventListener('click', () => {
            document.querySelectorAll('.filtro-btn')
                .forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
            filtroActual = btn.dataset.filtro;
            cargarMesas();
        });
    });
}

function initBuscador() {
    const input = document.getElementById('buscadorMesas');
    if (!input) return;
    input.addEventListener('input', e => {
        busqueda = e.target.value.trim();
        renderMesas();
    });
}

function actualizarContadores() {
    const c = {
        todas:      mesasData.length,
        abiertas:   mesasData.filter(m => m.estado === 'ocupada').length,
        pagadas:    mesasData.filter(m => m.estado === 'pagada').length,
        reservadas: mesasData.filter(m => m.estado === 'reservada').length,
    };
    const $ = id => document.getElementById(id);
    $('countTodas').textContent      = c.todas;
    $('countAbiertas').textContent   = c.abiertas;
    $('countPagadas').textContent    = c.pagadas;
    $('countReservadas').textContent = c.reservadas;
}

/* ---------- Modal de detalle ---------- */
function initModal() {
    const modal = document.getElementById('modalDetalle');
    document.getElementById('cerrarModal')
        .addEventListener('click', () => modal.classList.remove('open'));
    modal.addEventListener('click', e => {
        if (e.target === modal) modal.classList.remove('open');
    });
}

async function abrirDetalleMesa(id) {
    const modal  = document.getElementById('modalDetalle');
    const titulo = document.getElementById('modalTitulo');
    const body   = document.getElementById('modalBody');

    titulo.textContent = `Mesa #${id}`;
    body.innerHTML = '<p>Cargando...</p>';
    modal.classList.add('open');

    try {
        const consumo = await MesasAPI.detalle(id);
        renderDetalle(consumo);
    } catch (e) {
        // Demo
        renderDetalle(getConsumoDemo(id));
    }
}

function renderDetalle(consumo) {
    const body = document.getElementById('modalBody');
    const itemsHTML = consumo.items.map(it => `
        <div class="detalle-item">
            <span>${it.cantidad}× ${it.nombre}</span>
            <span>${Utils.formatearMoneda(it.precio * it.cantidad)}</span>
        </div>
    `).join('');

    body.innerHTML = `
        ${itemsHTML}
        <div class="detalle-total">
            <span>TOTAL</span>
            <span>${Utils.formatearMoneda(consumo.total)}</span>
        </div>
    `;
}

from flask import Blueprint, jsonify, request
from models import db
from models.pedido import Pedido, PedidoItem
from models.mesa import Mesa
from models.producto import Producto

pos_bp = Blueprint('pos', __name__)


@pos_bp.route('/productos', methods=['GET'])
def lista_productos():
    """Devuelve la carta activa."""
    prods = Producto.query.filter_by(activo=True).all()
    return jsonify([p.to_dict() for p in prods])


@pos_bp.route('', methods=['POST'])
def crear_pedido():
    """Crea un pedido y sus items."""
    data = request.get_json()
    mesa_id = data['mesa_id']

    pedido = Pedido(mesa_id=mesa_id, estado='pendiente')
    db.session.add(pedido)
    db.session.flush()  # para obtener pedido.id

    total = 0
    for item in data['items']:
        prod = Producto.query.get(item['producto_id'])
        if not prod:
            continue
        pi = PedidoItem(
            pedido_id=pedido.id,
            producto_id=prod.id,
            cantidad=item['cantidad'],
            precio_unitario=prod.precio,
            notas=item.get('notas', ''),
        )
        total += prod.precio * item['cantidad']
        db.session.add(pi)

    pedido.total = total

    # Marcar la mesa como ocupada
    mesa = Mesa.query.get(mesa_id)
    if mesa:
        mesa.estado = 'ocupada'

    db.session.commit()
    return jsonify(pedido.to_dict()), 201


@pos_bp.route('/<int:pedido_id>/enviar-cocina', methods=['POST'])
def enviar_cocina(pedido_id):
    """Marca el pedido como enviado a cocina."""
    pedido = Pedido.query.get_or_404(pedido_id)
    pedido.estado = 'en_cocina'
    db.session.commit()
    return jsonify({'ok': True, 'estado': pedido.estado})
